// Shimmering veil
// The veil (+50% CpS, more with Reinforced membrane & co.) breaks when you
// click the big cookie or pop a golden cookie / reindeer, and turning it
// back on costs 24 hours of unbuffed CpS.
//
// PROTECT_VEIL = true  -> idle play: while the veil is up the script does not
//                         click the big cookie, does not pop shimmers and does
//                         not cast Force the Hand of Fate. If the veil is
//                         already broken it plays actively (pops, clicks,
//                         combos) until buffs run out, then turns the veil
//                         back on.
// PROTECT_VEIL = false -> original active combo play, veil gets broken.
var PROTECT_VEIL = true;
// Turn the veil back on automatically once no buff is running and you
// can afford it with VEIL_REBUY_BANK_MULT times its price in the bank
// (2 = never spend more than half of your bank on it).
var AUTO_REENABLE_VEIL = false;
var VEIL_REBUY_BANK_MULT = 2;
// While the veil is up, only start clicking the big cookie during a click
// buff (Click frenzy, Dragonflight, ...) if the expected cookies from the
// remaining buff time exceed this many times the veil re-enable price.
var VEIL_BREAK_MARGIN = 3;
// Clicking the big cookie breaks the veil, so with PROTECT_VEIL the script
// would never click at all. With BREAK_VEIL_FOR_CLICKING on, it compares
// what non-stop clicking earns (CLICKER_PER_SECOND * cookies per click)
// with what the veil adds (veil boost * CpS). If clicking earns more than
// VEIL_CLICK_MARGIN times the veil, the veil is given up and the script
// plays actively (clicks, pops golden cookies, combos).
var BREAK_VEIL_FOR_CLICKING = true;
var VEIL_CLICK_MARGIN = 1;

// Big cookie clicks per second (browsers throttle background tabs to ~1/s)
const CLICKER_PER_SECOND = 10;

// Auto buy
// Buys every affordable upgrade (cheapest first) and then the building with
// the best payback time. Toggles (Elder Pledge/Covenant, veil, season
// switches, golden switch, ...), upgrades that cost sugar lumps, upgrades
// the game asks you to confirm (e.g. One mind) and the Chocolate egg are
// never bought.
var AUTO_BUY_UPGRADES = true;
var AUTO_BUY_BUILDINGS = true;
// While golden cookies are being popped, keep this many seconds of
// unbuffed CpS in the bank so Lucky! pays out in full (6000 s = max Lucky,
// 42000 s = max Lucky during a Frenzy). Ignored while the veil is protected.
var BANK_RESERVE_SECONDS = 6000;
// Never auto-buy these
var UPGRADE_BLACKLIST = ['Chocolate egg'];

// Grimoire
const MULTIPLIER_THRESHOLD_1 = 1000;
const MULTIPLIER_THRESHOLD_2 = 1000;
const CLICKER_THRESHOLD_1 = 300;
const CLICKER_THRESHOLD_2 = 300;
const SL_MULTIPLIER_THRESHOLD = 10000;
const SL_CLICKER_THRESHOLD = 1000;

// Garden
// Level 9+ farms already have the full 6x6 plot, so a level 11 farm uses every
// tile and all layouts below run unchanged.
var MAXIMIZE_SUGAR_LUMPS = true;
var PLANT_THRESHOLD = 5;
var BAKEBERRY_THRESHOLD_1 = 7;
var BAKEBERRY_THRESHOLD_2 = 50;

// Krumblor (dragon)
// Train the dragon automatically. Levels 5-24 sacrifice 100 of one building,
// then 50 and 200 of every building. Training only happens outside of buffs
// and, if DRAGON_REBUY_SACRIFICED is on, only once the bank holds
// DRAGON_TRAIN_BANK_MULT times the cost of buying the sacrificed buildings
// back (they are rebought right after).
var AUTO_TRAIN_DRAGON = true;
var DRAGON_REBUY_SACRIFICED = true;
var DRAGON_TRAIN_BANK_MULT = 2;
// Pick auras automatically. Every aura change sacrifices 1 of your best
// building (the game's own price), which the script buys back. Auras are
// only switched while no buff and no golden cookie is active. The first
// entry goes into slot 1, the second into slot 2 (needs a fully trained
// dragon). Unknown auras are skipped.
//   IDLE_AURAS   -> used while the Shimmering veil is protected. Golden
//                   cookies are left on screen there, so Dragon's Fortune
//                   (+123% CpS per golden cookie on screen) works for free.
//   ACTIVE_AURAS -> used when the script pops golden cookies and combos.
var AUTO_DRAGON_AURAS = true;
var IDLE_AURAS = ['Radiant Appetite', "Dragon's Fortune"];
var ACTIVE_AURAS = ['Radiant Appetite', 'Dragonflight'];
// Dragon Orbs: with no buffs and no golden cookie on screen, selling your
// best building has a 10% chance to spawn a golden cookie. If enabled, the
// active auras use Dragon Orbs instead of the 2nd ACTIVE_AURAS entry and
// the script sells + rebuys your best building to fish for golden cookies.
// One attempt loses ~62% of that building's price (~37% with Earth
// Shatterer), ~10 attempts per golden cookie on average. Attempts only run
// while one costs at most ORBS_MAX_COST_SECONDS of unbuffed CpS.
// Run dragon_orbs_report() in the console to see what it costs you.
var USE_DRAGON_ORBS = false;
var ORBS_MAX_COST_SECONDS = 30;
var ORBS_ATTEMPTS_PER_TICK = 10;


// ---------------------------------------------------------------------
// Shimmering veil helpers
// ---------------------------------------------------------------------

function veil_owned(){
    return Game.Has('Shimmering veil');
}

// 'Shimmering veil [off]' being owned means the veil is currently ON
function veil_up(){
    return veil_owned() && Game.Has('Shimmering veil [off]');
}

// True while the script should avoid anything that breaks the veil
function veil_protected(){
    return PROTECT_VEIL && veil_up() && !clicking_beats_veil();
}

// The script plays idle (veil) style as long as it wants to keep the veil
function playing_idle(){
    return PROTECT_VEIL && veil_owned() && !clicking_beats_veil();
}

function veil_price(){
    return Game.Upgrades['Shimmering veil [off]'].getPrice();
}

function any_buff_active(){
    for (var i in Game.buffs) {
        return true;
    }
    return false;
}

// Seconds left on the longest running click buff (0 if none)
function click_buff_seconds(){
    var seconds = 0;
    for (var i in Game.buffs) {
        var buff = Game.buffs[i];
        if (typeof buff.multClick !== 'undefined' && buff.multClick > 1) {
            seconds = Math.max(seconds, buff.time/Game.fps);
        }
    }
    return seconds;
}

// Is clicking out the current click buff worth breaking the veil for?
function click_worth_breaking_veil(){
    var seconds = click_buff_seconds();
    if (seconds <= 0) return false;
    var expected = seconds*CLICKER_PER_SECOND*Game.computedMouseCps;
    return expected > veil_price()*VEIL_BREAK_MARGIN;
}

// CpS multiplier bonus of the veil (0.5 = +50%)
function veil_boost(){
    return (typeof Game.getVeilBoost === 'function') ? Game.getVeilBoost() : 0.5;
}

// Does clicking non-stop earn more than keeping the veil up?
// With base CpS C and veil boost b: veil up -> C*(1+b) CpS, no clicks.
// Veil down -> C CpS + clicks worth click(C). Clicks scale with CpS just
// like the veil does, so this boils down to clicks > b * current CpS
// whether the veil is currently up or not.
// Click buffs are divided out: a Click frenzy alone is judged by
// click_worth_breaking_veil() instead.
function clicking_beats_veil(){
    if (!BREAK_VEIL_FOR_CLICKING) return false;
    var click_mult = 1;
    for (var i in Game.buffs) {
        var buff = Game.buffs[i];
        if (typeof buff.multClick !== 'undefined' && buff.multClick > 0) click_mult *= buff.multClick;
    }
    var clicking = CLICKER_PER_SECOND*Game.computedMouseCps/click_mult;
    return clicking > veil_boost()*Game.cookiesPs*VEIL_CLICK_MARGIN;
}

var autoVeil = setInterval(function() {
    if (!PROTECT_VEIL || !AUTO_REENABLE_VEIL || !veil_owned() || veil_up()) return;
    // Clicking would just break it again
    if (clicking_beats_veil()) return;
    // Let any running combo finish first
    if (any_buff_active() || Game.shimmers.length > 0) return;
    var veil = Game.Upgrades['Shimmering veil [off]'];
    if (veil.unlocked && !veil.bought && Game.cookies >= veil_price()*VEIL_REBUY_BANK_MULT) {
        veil.buy();
    }
}, 5000);

function highest_building(){
    var best = null;
    for (var i in Game.Objects) {
        if (Game.Objects[i].amount > 0) best = Game.Objects[i];
    }
    return best;
}

// Building.buy() sells instead when the store is in sell mode
function buy_building(building, amount){
    if (amount <= 0) return;
    var mode = Game.buyMode;
    Game.buyMode = 1;
    building.buy(amount);
    Game.buyMode = mode;
}

function golden_on_screen(){
    return Game.shimmerTypes['golden'].n > 0;
}

// [[building, amount], ...] the given dragon level sacrifices
function dragon_level_sacrifice(level){
    var last = Game.dragonLevels.length-1;
    if (level >= 5 && level < last-2) return [[Game.ObjectsById[level-5], 100]];
    var per = (level == last-2) ? 50 : (level == last-1) ? 200 : 0;
    var list = [];
    if (per > 0) {
        for (var i in Game.Objects) list.push([Game.Objects[i], per]);
    }
    return list;
}

// Cookies needed to buy back the top `amount` of a building
function rebuy_cost(building, amount){
    return building.getReverseSumPrice(amount)/building.getSellMultiplier();
}

// Same as Game.UpgradeDragon() but without popping up the dragon menu
function train_dragon(){
    var level = Game.dragonLevel;
    if (level >= Game.dragonLevels.length-1) return;
    if (any_buff_active()) return;
    var step = Game.dragonLevels[level];
    if (!step.cost()) return;
    var sacrifice = dragon_level_sacrifice(level);
    if (DRAGON_REBUY_SACRIFICED && sacrifice.length > 0) {
        var total = 0;
        for (var i = 0; i < sacrifice.length; i++) total += rebuy_cost(sacrifice[i][0], sacrifice[i][1]);
        if (Game.cookies < total*DRAGON_TRAIN_BANK_MULT) return;
    }
    step.buy();
    Game.dragonLevel = level+1;
    if (Game.dragonLevel >= Game.dragonLevels.length-1) Game.Win('Here be dragon');
    Game.recalculateGains = 1;
    Game.upgradesToRebuild = 1;
    console.log('Dragon trained: ' + Game.dragonLevels[level].action.replace(/<[^>]*>/g, ' '));
    if (DRAGON_REBUY_SACRIFICED) {
        for (var i = 0; i < sacrifice.length; i++) buy_building(sacrifice[i][0], sacrifice[i][1]);
    }
}

function aura_slots(){
    return Game.dragonLevel >= Game.dragonLevels.length-1 ? 2 : 1;
}

function aura_known(id){
    return id >= 0 && Game.dragonLevel >= id+4;
}

// Same as confirming the aura prompt: costs 1 of your best building
function set_aura(slot, id){
    var best = highest_building();
    if (slot == 0) Game.dragonAura = id; else Game.dragonAura2 = id;
    if (best) {
        best.sacrifice(1);
        buy_building(best, 1);
    }
    Game.recalculateGains = 1;
    console.log('Dragon aura ' + (slot+1) + ' set to ' + Game.dragonAuras[id].name);
}

function wanted_auras(){
    var names = veil_protected() ? IDLE_AURAS.slice() : ACTIVE_AURAS.slice();
    if (!veil_protected() && USE_DRAGON_ORBS) names[1] = 'Dragon Orbs';
    var ids = [];
    for (var i = 0; i < names.length && ids.length < aura_slots(); i++) {
        var aura = Game.dragonAurasBN[names[i]];
        if (!aura) { console.warn('Unknown dragon aura: ' + names[i]); continue; }
        if (aura_known(aura.id) && ids.indexOf(aura.id) == -1) ids.push(aura.id);
    }
    return ids;
}

function apply_dragon_auras(){
    if (any_buff_active() || golden_on_screen()) return;
    var wanted = wanted_auras();
    var current = [Game.dragonAura, Game.dragonAura2];
    for (var i = 0; i < wanted.length; i++) {
        if (current.slice(0, aura_slots()).indexOf(wanted[i]) != -1) continue;
        // Put it into a slot that holds an aura we don't want
        for (var slot = 0; slot < aura_slots(); slot++) {
            if (wanted.indexOf(current[slot]) == -1) {
                set_aura(slot, wanted[i]);
                current[slot] = wanted[i];
                break;
            }
        }
    }
}

var autoDragon = setInterval(function() {
    if (AUTO_TRAIN_DRAGON) train_dragon();
    if (AUTO_DRAGON_AURAS) apply_dragon_auras();
}, 5000);

// Cookies lost by selling 1 of the building and buying it back
function orbs_attempt_cost(building){
    var gain = Math.floor(building.getPrice()*building.getSellMultiplier());
    var rebuy = Math.ceil(Game.modifyBuildingPrice(building,
        building.basePrice*Math.pow(Game.priceIncrease, Math.max(0, building.amount-1-building.free))));
    return {gain: gain, rebuy: rebuy, loss: rebuy-gain};
}

function dragon_orbs_report(){
    var best = highest_building();
    if (!best) return console.log('No buildings owned.');
    var cost = orbs_attempt_cost(best);
    var seconds = cost.loss/Game.unbuffedCps;
    console.log('Dragon Orbs on ' + best.name + ' #' + best.amount + ': one attempt loses ' +
        Beautify(cost.loss) + ' cookies = ' + Beautify(seconds) + ' s of unbuffed CpS, ~' +
        Beautify(seconds*10) + ' s (' + Beautify(seconds/6) + ' min) per golden cookie. ' +
        'Limit: ' + ORBS_MAX_COST_SECONDS + ' s per attempt -> ' +
        (seconds <= ORBS_MAX_COST_SECONDS ? 'worth it.' : 'too expensive, script would skip it.'));
}

var autoDragonOrbs = setInterval(function() {
    if (!USE_DRAGON_ORBS || veil_protected()) return;
    if (Game.auraMult('Dragon Orbs') <= 0) return;
    // Godzamok gives a Devastation buff on every sale, which blocks the wish
    if (Game.hasGod && Game.hasGod('ruin')) return;
    var best = highest_building();
    if (!best) return;
    for (var k = 0; k < ORBS_ATTEMPTS_PER_TICK; k++) {
        if (golden_on_screen() || any_buff_active()) return;
        var cost = orbs_attempt_cost(best);
        if (cost.loss > Game.unbuffedCps*ORBS_MAX_COST_SECONDS) return;
        if (Game.cookies+cost.gain < cost.rebuy) return;
        best.sell(1);
        buy_building(best, 1);
    }
}, 1000);

// Buy Wizard towers back up to `target` after selling them for a cheap cast.
// Stops when the bank runs dry instead of looping forever.
function rebuy_towers(target){
    var tower = Game.ObjectsById[7];
    while (tower.amount < target && Game.cookies >= tower.getPrice()) {
        var before = tower.amount;
        buy_building(tower, 1);
        if (tower.amount == before) break;
    }
}

// True while a CpS or click buff runs (the Grimoire combos sell and rebuy
// Wizard towers then, so don't spend their cookies). Long harmless buffs
// such as Sugar blessing don't count.
function combo_running(){
    for (var i in Game.buffs) {
        var buff = Game.buffs[i];
        if ((buff.multCpS && buff.multCpS > 1) || (buff.multClick && buff.multClick > 1)) return true;
    }
    return false;
}

function bank_reserve(){
    var reserve = 0;
    if (!veil_protected()) reserve = Game.unbuffedCps*BANK_RESERVE_SECONDS;
    // Keep the veil re-enable money if the script is going to turn it back on
    if (PROTECT_VEIL && AUTO_REENABLE_VEIL && veil_owned() && !veil_up() && !clicking_beats_veil()) {
        reserve = Math.max(reserve, veil_price()*VEIL_REBUY_BANK_MULT);
    }
    return reserve;
}

function upgrade_skipped(u){
    if (u.bought || u.pool == 'toggle' || u.pool == 'debug' || u.pool == 'prestige') return true;
    if (u.priceLumps) return true;
    // Upgrades that ask for confirmation (One mind, ...) are left to you
    if (u.clickFunction || u.choicesFunction) return true;
    if (UPGRADE_BLACKLIST.indexOf(u.name) != -1) return true;
    if (u.name.indexOf('Shimmering veil') == 0) return true;
    for (var s in Game.seasons) {
        if (Game.seasons[s].trigger == u.name) return true;
    }
    return false;
}

// Cheapest affordable upgrade, or null
function next_upgrade(budget){
    var best = null;
    for (var i = 0; i < Game.UpgradesInStore.length; i++) {
        var u = Game.UpgradesInStore[i];
        if (upgrade_skipped(u)) continue;
        if (u.canBuy && !u.canBuy()) continue;
        var price = u.getPrice();
        if (price > budget) continue;
        if (!best || price < best.getPrice()) best = u;
    }
    return best;
}

// Building with the best payback period (Cookie Monster style: time to
// afford + time to earn back its price). Synergies are ignored.
function next_building(budget){
    var best = null, best_pp = Infinity;
    var cps = Math.max(Game.cookiesPs, 1);
    for (var i in Game.Objects) {
        var b = Game.Objects[i];
        if (b.locked) continue;
        var gain = b.storedCps*Game.globalCpsMult;
        if (!(gain > 0)) continue;
        var price = b.getPrice();
        var pp = Math.max(price-budget, 0)/cps + price/gain;
        if (pp < best_pp) { best_pp = pp; best = b; }
    }
    // Only buy the best one; if it's not affordable yet, save up for it
    if (best && best.getPrice() <= budget) return best;
    return null;
}

var autoBuy = setInterval(function() {
    if (combo_running()) return;
    for (var k = 0; k < 100; k++) {
        var budget = Game.cookies-bank_reserve();
        if (budget <= 0) return;
        if (AUTO_BUY_UPGRADES) {
            var u = next_upgrade(budget);
            if (u) {
                u.buy(true);
                if (!u.bought) return;
                Game.CalculateGains();
                continue;
            }
        }
        if (AUTO_BUY_BUILDINGS) {
            var b = next_building(budget);
            if (b) {
                var before = b.amount;
                buy_building(b, 1);
                if (b.amount == before) return;
                Game.CalculateGains();
                continue;
            }
        }
        return;
    }
}, 2000);

function farm_bakeberries() {
    var farm = Game.ObjectsById[2].minigame;
    if(farm.plantsById[8].unlocked == 1){
        for (let i = 0; i < 6; i++) {
            for (let j = 0; j < 6; j++) {
                var tile = farm.plot[j][i]
                if (tile[0] == 0){
                    if (Game.cookiesPs/Game.unbuffedCps < PLANT_THRESHOLD) {
                        farm.seedSelected = 8
                        farm.clickTile(i,j);
                    }
                }
                else if (tile[1] > 90){
                    if (Game.cookiesPs/Game.unbuffedCps > BAKEBERRY_THRESHOLD_1) {
                        farm.clickTile(i,j);
                    }
                }
                else if (tile[1] >= farm.plantsById[tile[0]-1].mature){
                    if (Game.cookiesPs/Game.unbuffedCps > BAKEBERRY_THRESHOLD_2) {
                        farm.clickTile(i,j);
                    }
                }
            }
        }
    }
}

// Golden clovers only help if golden cookies get popped (combo play)
function farm_goldenclover() {
    var farm = Game.ObjectsById[2].minigame;
    change_soil(farm,2);
    if(farm.plantsById[5].unlocked == 1){
        for (let i = 0; i < 6; i++) {
            for (let j = 0; j < 6; j++) {
                var tile = farm.plot[j][i]
                if (tile[0] == 0){
                    if (Game.cookiesPs/Game.unbuffedCps < PLANT_THRESHOLD) {
                        farm.seedSelected = 5
                        farm.clickTile(i,j);
                    }
                }
                else if (tile[0] != 6){
                    farm.clickTile(i,j);
                }
            }
        }
    }
}

// Veil (idle) play: fill all 36 tiles with Baker's wheat on clay.
// Each one gives +1% CpS (x1.25 on clay -> about +45% CpS total) for only
// one minute of CpS per seed. Dead wheat is simply replanted and any
// weeds / mutations that sprout get pulled.
function farm_veil_idle() {
    var farm = Game.ObjectsById[2].minigame;
    var wheat = plant_mapping['bakerWheat'];
    change_soil(farm,2);
    for (let i = 0; i < 6; i++) {
        for (let j = 0; j < 6; j++) {
            if (!farm.isTileUnlocked(i,j)) continue;
            var tile = farm.plot[j][i]
            if (tile[0] == 0){
                plant_tiles(farm, wheat, [i], [j]);
            }
            else if (tile[0]-1 != wheat && farm.plantsById[tile[0]-1].unlocked == 1){
                farm.clickTile(i,j);
            }
        }
    }
}

function farm_endgame() {
    if (playing_idle()) {
        farm_veil_idle();
    }
    else {
        farm_goldenclover();
    }
}




// Helper functions

function all_seeds_unlocked(){
    var farm = Game.ObjectsById[2].minigame;
    // iterate through all seeds in dictionary farm.plants
    for (var i in farm.plants) {
        if (farm.plants[i].unlocked == 0){
            return false;
        }
    }
    return true;
}

function check_tiles(farm, plantId, x_axis, y_axis){
    for (let i = 0; i < x_axis.length; i++) {
        for (let j = 0; j < y_axis.length; j++) {
            var tile = farm.plot[y_axis[j]][x_axis[i]]
            if (tile[0] == 0 || tile[0]-1 != plantId && farm.plantsById[tile[0]-1].unlocked == 1){
                return false;
            }
        }
    }
    return true;
};

function plant_tiles(farm, plantId, x_axis, y_axis){
    for (let i = 0; i < x_axis.length; i++) {
        for (let j = 0; j < y_axis.length; j++) {
            if (Game.cookiesPs/Game.unbuffedCps < PLANT_THRESHOLD) {
                farm.seedSelected = plantId;
                // If plant at time is unlocked or tile is empty plant it
                var tile = farm.plot[y_axis[j]][x_axis[i]]
                if (tile[0] == 0 || farm.plantsById[tile[0]-1].unlocked == 1){
                    farm.clickTile(x_axis[i],y_axis[j]);
                }
            }
        }
    }
    return true;
};

function change_soil(farm, soilId){
    if (!(farm.freeze || farm.soil==soilId || farm.nextSoil>Date.now())){
        farm.nextSoil=Date.now()+(Game.Has('Turbo-charged soil')?1:(1000*60*10));
        farm.toCompute=true;farm.soil=soilId;farm.computeStepT();
        for (var i in farm.soils){var it=farm.soils[i];if (it.id==farm.soil){l('gardenSoil-'+it.id).classList.add('on');}else{l('gardenSoil-'+it.id).classList.remove('on');}}
    }
};

function any_mature(farm){
    var x_axis = [0,1,2,3,4,5];
    var y_axis = [0,1,2,3,4,5];
    for (let i = 0; i < x_axis.length; i++) {
        for (let j = 0; j < y_axis.length; j++) {
            var tile = farm.plot[y_axis[j]][x_axis[i]]
            if (tile[0] != 0 && farm.plantsById[tile[0]-1].mature <= tile[1]){
                return true;
            }
        }
    }
    return false;
};

function get_mature_time(parent, age) {
    var farm = Game.ObjectsById[2].minigame;
    var rate = farm.plantsById[parent].ageTick;
    var ticks2mature = farm.plantsById[parent].mature - age;
    return ticks2mature/rate;
};

function get_long_short(parent1, parent2) {
    parent1 = plant_mapping[parent1]
    parent2 = plant_mapping[parent2]
    if (get_mature_time(parent1, 0) > get_mature_time(parent2, 0)){
        return [parent1, parent2]
    }
    else{
        return [parent2, parent1]
    }
};

function spawn() {
    var farm = Game.ObjectsById[2].minigame;
    change_soil(farm,1);
    for (let i = 0; i < 6; i++) {
        for (let j = 0; j < 6; j++) {
            var tile = farm.plot[j][i]
            if (tile[0] != 0 && tile[1] >= farm.plantsById[tile[0]-1].mature && (farm.plantsById[tile[0]-1].unlocked == 0 || tile[0]-1 == plant_mapping['meddleweed'])){
                farm.clickTile(i,j);
            }
            if (tile[0] != 0 && tile[0]-1 != plant_mapping['meddleweed'] && farm.plantsById[tile[0]-1].unlocked != 0){
                farm.clickTile(i,j);
            }
        }
    }
}

function clean_meddle(){
    var farm = Game.ObjectsById[2].minigame;
    change_soil(farm,1);
    for (let i = 0; i < 6; i++) {
        for (let j = 0; j < 6; j++) {
            var tile = farm.plot[j][i]
            if (tile[0] != 0 && (tile[1] >= farm.plantsById[tile[0]-1].mature || (tile[0]-1 != plant_mapping['meddleweed'] && farm.plantsById[tile[0]-1].unlocked == 1))){
                farm.clickTile(i,j);
            }
        }
    }

}

function plant_meddle(){
    var farm = Game.ObjectsById[2].minigame;
    change_soil(farm,1);
    for (let i = 0; i < 6; i++) {
        for (let j = 0; j < 6; j++) {
            var tile = farm.plot[j][i];
            if (tile[0] == 0){
                plant_tiles(farm, plant_mapping['meddleweed'], [i], [j])
            }
        }
    }

}

function meddleweed() {
    clean_meddle();
    plant_meddle();
}

function maximize_soil(farm) {
    if(any_mature(farm)){// && all_unlocked(farm)){
        change_soil(farm,4);
    }
    else{
        change_soil(farm,1);
    }
}

function breed_self(parent) {
    var farm = Game.ObjectsById[2].minigame;
    maximize_soil(farm)
    parent = plant_mapping[parent];
    if (!check_self_mutition(farm, parent)){ 
        clean_all_plants(farm);
        plant_self_mutition(farm, parent);
    }
    clean_self_mutition(farm)
}

function breed_cross(parent1, parent2) {
    var farm = Game.ObjectsById[2].minigame;
    maximize_soil(farm)
    new_ordering = get_long_short(parent1, parent2)
    longer = new_ordering[0]
    shorter = new_ordering[1]
    if (!check_longer_plant(farm, longer)){ 
        clean_all_plants(farm);
        plant_longer_mutition(farm, longer);
    }
    if(!check_shorter_plant(farm, shorter)){
        clean_short_mutations(farm);
        plant_shorter_mutition(farm, shorter)
    }
    clean_two_mutition(farm)
}

// Plants breed self or breed cross along the edges of the 8x8 farm
function side_plant_eight_mutition(farm){
    // Select plant that is not unlocked
    for (var plant in plant_breeding) {
        if(farm.plants[plant].unlocked == 0 && !plant_growing(farm, plant) && parents_unlocked(farm, plant_breeding[plant]) && (plant_breeding[plant].method == 'Self' || plant_breeding[plant].method == 'Cross') && plant != "duketater"){
            if(plant_breeding[plant].method == 'Self'){
                side_breed(plant_breeding[plant].parent, plant_breeding[plant].parent);
            }
            else if(plant_breeding[plant].method == 'Cross'){
                side_breed(plant_breeding[plant].parent1, plant_breeding[plant].parent2);
            }
            break;
        }
    }

}

// Checks if a plant with a given name is currently in the farm
function plant_growing(farm, plant){
    plant = plant_mapping[plant];
    for (let i = 0; i < 6; i++) {
        for (let j = 0; j < 6; j++){
            var tile = farm.plot[j][i];
            if (tile[0]-1 == plant){
                return true;
            }
        }
    }
    return false;
}

function parents_unlocked(farm, parents){
    if (parents.method == 'Spawn' || parents.method == 'meddleweed'){
        return true;
    }
    else if (parents.method == 'Self' || parents.method == 'Three' || parents.method == 'Eight'){
        if (farm.plants[parents.parent].unlocked == 1){
            return true;
        }
    }
    else{
        if(farm.plants[parents.parent1].unlocked == 1 && farm.plants[parents.parent2].unlocked == 1){
            return true;
        }
    }

}

var get_all_plants = setInterval(function() {
    var farm = Game.ObjectsById[2].minigame;
    // iterate through plant_breeding
    var growing = false;
    
    for (var plant in plant_breeding) {
        if(farm.plants[plant].unlocked == 0 && !plant_growing(farm, plant) && parents_unlocked(farm, plant_breeding[plant])){
            growing = true;
            if(plant_breeding[plant].method == 'Spawn'){
                spawn();
            }
            if(plant_breeding[plant].method == 'meddleweed'){
                meddleweed();
            }
            else if(plant_breeding[plant].method == 'Self'){
                breed_self(plant_breeding[plant].parent);
            }
            else if(plant_breeding[plant].method == 'Cross'){
                breed_cross(plant_breeding[plant].parent1, plant_breeding[plant].parent2);
            }
            else if(plant_breeding[plant].method == 'Three'){
                breed_three(plant_breeding[plant].parent);
            }
            else if(plant_breeding[plant].method == 'Eight'){
                breed_eight(plant_breeding[plant].parent);
            }
            else if(plant_breeding[plant].method == '3x3'){
                breed_three_by_three(plant_breeding[plant].parent1, plant_breeding[plant].parent2);
            }
            break;
        }
    }
    if (!all_seeds_unlocked()){
        harvest_unlocked(farm);
    }
    
    if(!growing && MAXIMIZE_SUGAR_LUMPS && !all_seeds_unlocked()){
        clean_all_plants(farm);
        maximize_soil(farm);
    }
    if(all_seeds_unlocked()){
        if(MAXIMIZE_SUGAR_LUMPS && Game.lumps <= 100){
            farm.convert();
        }
        else{
            farm_endgame();
            return;
        }
    }
}, 10000);
